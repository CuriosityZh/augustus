cmake_minimum_required(VERSION 3.16)
project(augustus_planning_tests C)
set(CMAKE_C_STANDARD 99)
option(PLANNING_SANITIZERS "Enable address and undefined behavior checks" ON)
enable_testing()
add_executable(planning-tests test_model.c ../../src/planning/model.c ../../src/planning/storage.c ../../src/planning/sha256.c)
target_include_directories(planning-tests PRIVATE ../../src)
target_compile_options(planning-tests PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
if(PLANNING_SANITIZERS AND CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(planning-tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(planning-tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME planning-model-and-storage COMMAND planning-tests)
